cmake_minimum_required(VERSION 3.16)
option(FA_REFERENCE_ONLY "Build the FP64 host reference only (no NPU)" OFF)
if(FA_REFERENCE_ONLY)
    project(flash_attention_fixed LANGUAGES CXX)
else()
    set(CMAKE_ASC_ARCHITECTURES dav-2201 CACHE STRING "Only supported target")
    if(NOT CMAKE_ASC_ARCHITECTURES STREQUAL "dav-2201")
        message(FATAL_ERROR "This fixed example targets dav-2201 only")
    endif()
    find_package(ASC REQUIRED)
    project(flash_attention_fixed LANGUAGES ASC CXX)
    set_source_files_properties(main.cpp PROPERTIES LANGUAGE ASC)
endif()
add_executable(fa_fixed main.cpp)
set_target_properties(fa_fixed PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
if(FA_REFERENCE_ONLY)
    target_compile_definitions(fa_fixed PRIVATE FA_REFERENCE_ONLY=1)
else()
    target_compile_options(fa_fixed PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201> $<$<COMPILE_LANGUAGE:ASC>:-std=c++17>)
    target_link_directories(fa_fixed PRIVATE "$ENV{ASCEND_HOME_PATH}/lib64")
    target_link_libraries(fa_fixed PRIVATE tiling_api register platform m dl)
endif()
